[@:逐字字符串-引號里的全為字符，包括空格/換行]

string str = "C:\\Windows";
[\:轉義符 將\原文輸出]

[相當于：我说"你好"，双引号特殊，要2个]


int a = 10;
&a：房子的门牌号码（内存地址）可能是--地址1000 → [10]
int* p = &a; [数据类型 * ——指针]
  • 将p指向a的内存位置[1000]
  • p：售楼中介p找到房子主人a的房子
  • *p：售楼中介取得房子a[&a]待售

—————————————————————
void change(int* p)  //改一个内存地址
{
  *p = 100;  //任何扔进去的数值都变为100
}

int a = 10;
change(&a);  //把a的内存地址扔进去更改
